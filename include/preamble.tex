% theo: top-ruled box with a detached title, from Bertalan's BPM paper.
% Usage: \begin{tcolorbox}[theo=blue, title={Key insight}] ... \end{tcolorbox}

    theo/.style={%
        enhanced,

        toprule=1pt, rightrule=0pt, bottomrule=0pt, leftrule=0pt,

            \node[minimum width=1cm, anchor=south west, font=\bfseries]
               at (frame.north west) {\tcbtitle};
        }
    }
}

% The environments above share the theorem counter, so cleveref would call all of them
% "Theorem". Inside each one, labels on that counter get the environment's own name.
\makeatletter
\@for\tdk@env:=lemma,proposition,corollary,invariant,property,claim,remark,notation\do{%
  \edef\tdk@hook{\noexpand\AddToHook{env/\tdk@env/begin}{\noexpand\crefalias{theorem}{\tdk@env}}}%
  \tdk@hook}
\makeatother

% Tagged lists: numbered items with a short tag that \cref can point to.
%   \NewTagList{<environment>}{<tag>}{<Name>}{<Names>}
% defines an enumerate-like environment whose items read "C1", "C2", ... and whose
% references read "Contribution C1", "Contributions C1 and C2".
% Usage:
%   \begin{contributions}
%     \item\label{c:model} A formal model of ...
%   \end{contributions}
%   ... as \cref{c:model} shows ...
\newcommand{\NewTagList}[4]{%
  \newlist{#1}{enumerate}{1}%
  \setlist[#1]{label=\textbf{#2\arabic*}, ref=#2\arabic*, leftmargin=*, widest=99}%
  \crefalias{#1i}{#1}%
  \crefname{#1}{#3}{#4}%
}
\NewTagList{contributions}{C}{Contribution}{Contributions}
\NewTagList{requirements}{R}{Requirement}{Requirements}
\NewTagList{assumptions}{A}{Assumption}{Assumptions}
\NewTagList{guarantees}{G}{Guarantee}{Guarantees}
\NewTagList{threats}{T}{Threat}{Threats}
\NewTagList{limitations}{L}{Limitation}{Limitations}
