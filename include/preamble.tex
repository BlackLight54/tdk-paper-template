% Tagged lists: numbered items with a short tag that \cref can point to.
%   \NewTagList{<environment>}{<tag>}{<Name>}{<Names>}
% defines an enumerate-like environment whose items read "AS1", "AS2", ... Each item starts
% with a short name in bold, which the gap matrix and the text reuse:
%   \begin{assumptions}
%     \item\label{asm:clock} \textbf{Bounded drift.} Clocks drift by at most $\delta$.
%   \end{assumptions}
% A reference prints the tag alone, as a student would type it: \cref{asm:clock} -> "AS1",
% \cref{asm:clock,asm:keys} -> "AS1 and AS2", three or more in a row -> "AS1–AS3". The
% tag says what the item is, so the sentence does not need to. <Name> is what \namecref prints.
% \tagrow{<environment>} gives a table row the list's next tag, for items kept in a table
% instead of a list: \tagrow{guarantees}\label{g:monotone} & Monotone growth & ...
% (\tdk@tagtype makes each kind's format differ, or cleveref would merge two kinds into one
% range: "AS1–SG3".)
\makeatletter
\newcommand{\tdk@tagtype}[1]{}
\newcommand{\NewTagList}[4]{%
  \newlist{#1}{enumerate}{1}%
  \setlist[#1]{label=\textbf{#2\arabic*}, ref=#2\arabic*, leftmargin=*, widest=99}%
  \expandafter\def\csname the#1i\endcsname{#2\arabic{#1i}}%
  \crefalias{#1i}{#1}%
  \crefname{#1}{#3}{#4}\Crefname{#1}{#3}{#4}%
  \crefformat{#1}{\tdk@tagtype{#1}##2##1##3}\Crefformat{#1}{\tdk@tagtype{#1}##2##1##3}%
  \crefmultiformat{#1}{\tdk@tagtype{#1}##2##1##3}{\tdk@tagtype{#1} and~##2##1##3}{\tdk@tagtype{#1}, ##2##1##3}{\tdk@tagtype{#1} and~##2##1##3}%
  \Crefmultiformat{#1}{\tdk@tagtype{#1}##2##1##3}{\tdk@tagtype{#1} and~##2##1##3}{\tdk@tagtype{#1}, ##2##1##3}{\tdk@tagtype{#1} and~##2##1##3}%
  \crefrangeformat{#1}{\tdk@tagtype{#1}##3##1##4--##5##2##6}\Crefrangeformat{#1}{\tdk@tagtype{#1}##3##1##4--##5##2##6}%
